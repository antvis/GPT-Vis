A theme you can take anywhere

Download the theme JSON beside your code, then pass it to any GPT-Vis chart.

import { GPTVis, type ThemeSpec } from '@antv/gpt-vis';
import theme from './gpt-vis.json';

const chart = new GPTVis({
  container: '#chart',
  theme: theme as ThemeSpec,
});

chart.render({
  type: 'line',
  data: [
    { time: 'Q1', value: 32 },
    { time: 'Q2', value: 48 },
    { time: 'Q3', value: 61 },
  ],
});
